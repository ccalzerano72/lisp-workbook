% =============================================================

\begin{envbox}[Avvio e REPL]{envBlue}

\small\color{black!70}
Avvia \fn{allegro-ansi.exe} \color{black!55}(modalità ANSI: i simboli diventano maiuscoli)\color{black!70}.
Compare il \textbf{\textcolor{envBlue!80!black}{prompt REPL}}:\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(1):
\end{lstlisting}

\small\color{black!70}
Digita un'espressione e premi \textbf{Invio}:\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(1): (+ 1 2)
3
CL-USER(2): (defun quadrato (x) (* x x))
QUADRATO
CL-USER(3): (quadrato 5)
25
\end{lstlisting}

\small\color{black!70}
\textbf{\textcolor{envBlue!80!black}{Riutilizzare risultati:}} \fn{*}, \fn{**}, \fn{***} = ultimi tre valori:\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(4): (+ 10 20)
30
CL-USER(5): (1+ *)       ; * vale 30 -> 31
31
\end{lstlisting}

\end{envbox}

% =============================================================
%  CARICARE E COMPILARE FILE
% =============================================================

\begin{envbox}[Caricare e compilare file]{envGreen}

\small\color{black!70}
Scrivi il codice in un file \fn{.lisp}, poi caricalo nel REPL:
\color{black}

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}l@{\quad}l@{}}
\rowcolor{envGreen!12}
\textcolor{envGreen!80!black}{\textbf{Comando}} & \textcolor{envGreen!80!black}{\textbf{Effetto}} \\
\fn{:ld "file.lisp"} & \small\color{black!65} Carica e valuta (interpretato) \\
\rowcolor{envGreen!5}
\fn{:cl "file.lisp"} & \small\color{black!65} Compila e carica (genera \fn{.fasl}) \\
\fn{:cf "file.lisp"} & \small\color{black!65} Solo compila \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\smallskip
\small\color{black!70}
\textbf{\textcolor{envGreen!80!black}{Gestione directory:}}\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(1): :pwd                   ; directory corrente
CL-USER(2): :cd "c:/esercizi-lisp" ; cambia directory
CL-USER(3): :ld "es01.lisp"        ; carica file
\end{lstlisting}

\begin{warnbox}[Percorsi Windows]
Usa \fn{/} nei percorsi, non backslash. Esempio: \fn{"c:/cartella/file.lisp"}
\end{warnbox}

\end{envbox}

\small\color{black!70}
I comandi \fn{:} sono specifici di Allegro \color{black!55}(non ANSI standard)\color{black!70}.
\color{black}

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}l@{\quad}l@{\quad}l@{}}
\rowcolor{envBlue!12}
\textcolor{envBlue!80!black}{\textbf{Comando}} & \textcolor{envBlue!80!black}{\textbf{Abbr.}} & \textcolor{envBlue!80!black}{\textbf{Effetto}} \\
\fn{:ld "file"} & --- & \small\color{black!65} carica file \\
\rowcolor{envBlue!5}
\fn{:cl "file"} & --- & \small\color{black!65} compila e carica \\
\fn{:pwd} / \fn{:cd "dir"} & \fn{:pw} & \small\color{black!65} directory \\
\rowcolor{envBlue!5}
\fn{:trace fn} & \fn{:tr} & \small\color{black!65} attiva trace \\
\fn{:untrace fn} & \fn{:untr} & \small\color{black!65} disattiva trace \\
\rowcolor{envBlue!5}
\fn{:inspect obj} & \fn{:i} & \small\color{black!65} ispeziona oggetto \\
\fn{:describe obj} & \fn{:de} & \small\color{black!65} documentazione \\
\rowcolor{envBlue!5}
\fn{:apropos "str"} & \fn{:ap} & \small\color{black!65} cerca simboli \\
\fn{:history} & \fn{:his} & \small\color{black!65} cronologia \\
\rowcolor{envBlue!5}
\fn{:help} & \fn{:he} & \small\color{black!65} lista comandi \\
\fn{:exit} & \fn{:ex} & \small\color{black!65} esci \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\smallskip
\small\color{black!70}
\textbf{\textcolor{envBlue!80!black}{Cercare funzioni}} con \fn{:apropos}:\nopagebreak
\color{black}

\begin{lstlisting}

REMOVE           function
REMOVE-IF        function
REMOVE-DUPLICATES function
\end{lstlisting}

\end{envbox}

% =============================================================
%  TRACE
% =============================================================

\begin{envbox}[Trace: seguire la ricorsione]{envOrange}

\small\color{black!70}
\fn{:trace} mostra ogni \emph{chiamata} e il \emph{valore restituito}:\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(1): (defun fatt (n)
              (if (= n 0) 1 (* n (fatt (1- n)))))
FATT
CL-USER(2): :trace fatt
CL-USER(3): (fatt 4)
  0: (FATT 4)
    1: (FATT 3)
      2: (FATT 2)
        3: (FATT 1)
          4: (FATT 0)
          4: FATT returned 1
        3: FATT returned 1
      2: FATT returned 2
    1: FATT returned 6
  0: FATT returned 24
24
\end{lstlisting}

\small\color{black!70}
Disattivare: \fn{:untrace fatt} \color{black!55}(o \fn{:untrace} per tutte)\color{black!70}.
\color{black}

\end{envbox}

\begin{envbox}[Il debugger]{envOrange}

\small\color{black!70}
In caso di \emph{errore}, Allegro entra nel \textbf{\textcolor{envOrange!80!black}{debugger}} (prompt \fn{[1]}):\nopagebreak
\color{black}

\begin{lstlisting}
CL-USER(1): (/ 1 0)
Error: Division-by-zero.
[1] CL-USER(2):
\end{lstlisting}

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}l@{\quad}l@{\quad}l@{}}
\rowcolor{envOrange!12}
\textcolor{envOrange!80!black}{\textbf{Comando}} & \textcolor{envOrange!80!black}{\textbf{Abbr.}} & \textcolor{envOrange!80!black}{\textbf{Effetto}} \\
\fn{:reset} & \fn{:res} & \small\color{black!65} torna al livello 0 \\
\rowcolor{envOrange!5}
\fn{:pop} & --- & \small\color{black!65} sale di un livello \\
\fn{:bt} / \fn{:zoom} & \fn{:zo} & \small\color{black!65} stampa stack \\
\rowcolor{envOrange!5}
\fn{:error} & \fn{:err} & \small\color{black!65} ristampa errore \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\begin{warnbox}[Livelli annidati]
Prompt \fn{[2]}+ = errori accumulati. Usa \fn{:reset} per tornare al livello 0.
\end{warnbox}

\end{envbox}

% =============================================================
%  ERRORI COMUNI
% =============================================================

\begin{envbox}[Errori comuni]{envOrange}

\small\color{black!70}
\textbf{\textcolor{envOrange!80!black}{Parentesi mancante:}} se Invio non fa nulla, manca una \fn{)}.
Chiudila o premi \textbf{Ctrl+C}.
\color{black}

\smallskip
\small\color{black!70}
\textbf{\textcolor{envOrange!80!black}{Parentesi in più:}}\nopagebreak
\color{black}
\begin{lstlisting}
CL-USER(1): (+ 1 2))
Error: eof encountered ...
\end{lstlisting}
\small\color{black!70}
Usa \fn{:reset} e riscrivi.
\color{black}

\smallskip
\small\color{black!70}
\textbf{\textcolor{envOrange!80!black}{Funzione non definita:}}\nopagebreak
\color{black}
\begin{lstlisting}
CL-USER(1): (mia-funzione 3)
Error: Undefined function: MIA-FUNZIONE
\end{lstlisting}
\small\color{black!70}
Verifica di aver caricato il file con \fn{:ld}.
\color{black}

\end{envbox}

\begin{envbox}[Flusso di lavoro consigliato]{envPurple}

\small\color{black!70}
\begin{enumerate}[leftmargin=1.5em, itemsep=3pt, topsep=0pt]
  \item Crea \fn{es01.lisp} con un editor
  \item Scrivi \textbf{\textcolor{envPurple!80!black}{una sola funzione}} per volta
  \item Nel REPL: \fn{:cd} + \fn{:ld "es01.lisp"}
  \item \textbf{Testa subito} con vari input
  \item Bug? Modifica, salva, ricarica con \fn{:ld}
  \item Per ricorsione: usa \fn{:trace}
  \item Passa alla prossima \emph{solo quando funziona}
\end{enumerate}
\color{black}

\begin{tipbox}[Sviluppo incrementale]
Non scrivere tutto insieme. \textbf{Scrivi poco}, \textbf{testa subito}, \textbf{procedi}.
\end{tipbox}

\end{envbox}
