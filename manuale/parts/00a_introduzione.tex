\section*{Cosa rende Lisp diverso}
% =============================================================

\begin{introbox}[Principi fondamentali]{csBlue}

\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}l@{\quad}l@{}}
\textcolor{csBlue!80!black}{\textbf{Codice = dati}} & \small\color{black!70} I programmi sono liste. Puoi scrivere programmi che generano programmi.\\
\textcolor{csBlue!80!black}{\textbf{Valutazione}} & \small\color{black!70} Non si "eseguono istruzioni": si \emph{valutano espressioni} che restituiscono valori.\\
\textcolor{csBlue!80!black}{\textbf{Funzioni first-class}} & \small\color{black!70} Le funzioni sono valori: si passano, si restituiscono, si memorizzano.\\
\textcolor{csBlue!80!black}{\textbf{Composizione}} & \small\color{black!70} Si costruisce componendo funzioni piccole, non accumulando passi.\\
\textcolor{csBlue!80!black}{\textbf{Sintassi minima}} & \small\color{black!70} Poche regole, molte convenzioni. Le parentesi sono struttura, non rumore.\\
\textcolor{csBlue!80!black}{\textbf{REPL}} & \small\color{black!70} Sviluppo interattivo: esplora, testa, costruisci incrementalmente.\\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\end{introbox}

\begin{introbox}[Imperativo vs Funzionale]{csOrange}

\renewcommand{\arraystretch}{1.3}
\centering
\begin{tabular}{@{}p{6.2cm}p{6.2cm}@{}}
\rowcolor{csOrange!12}
\textcolor{csOrange!80!black}{\textbf{Imperativo (C, Java, Python)}} & \textcolor{csOrange!80!black}{\textbf{Lisp / Funzionale}} \\
\small\color{black!70} Istruzioni che modificano stato & \small\color{black!70} Espressioni che producono valori \\
\rowcolor{csOrange!5}
\small\color{black!70} Variabili = contenitori mutabili & \small\color{black!70} \emph{Binding} = nomi associati a valori \\
\small\color{black!70} Cicli (\fn{for}, \fn{while}) & \small\color{black!70} Ricorsione e higher-order \\
\rowcolor{csOrange!5}
\small\color{black!70} Strutture di controllo fisse & \small\color{black!70} \emph{Macro}: controllo definibile \\
\small\color{black!70} Sintassi specifica per costrutto & \small\color{black!70} Sintassi uniforme (S-expression) \\
\rowcolor{csOrange!5}
\small\color{black!70} Compilazione separata & \small\color{black!70} Sviluppo interattivo nel REPL \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\end{introbox}

\begin{introbox}[Concetti chiave]{csGreen}

\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}l@{\quad}l@{}}
\textcolor{csGreen!80!black}{\textbf{S-expression}} & \small\color{black!70} Atomo o lista di S-expression. \emph{Tutto} il codice ha questa forma.\\
\textcolor{csGreen!80!black}{\textbf{Valutazione}} & \small\color{black!70} Numeri/stringhe → se stessi. Simboli → valore. Liste → chiamata funzione.\\
\textcolor{csGreen!80!black}{\textbf{Ambiente}} & \small\color{black!70} I binding visibili in un punto. \fn{let} crea ambienti locali.\\
\textcolor{csGreen!80!black}{\textbf{Ricorsione}} & \small\color{black!70} Il modo naturale per elaborare liste e alberi.\\
\textcolor{csGreen!80!black}{\textbf{Higher-order}} & \small\color{black!70} Funzioni che accettano/restituiscono funzioni: \fn{mapcar}, \fn{reduce}.\\
\textcolor{csGreen!80!black}{\textbf{Macro}} & \small\color{black!70} Trasformazioni di codice a compile-time. Estendono il linguaggio.\\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\end{introbox}

\begin{introbox}[Per chi viene da altri linguaggi]{csPurple}

\small\color{black!70}
\begin{itemize}[leftmargin=1.2em, itemsep=4pt, topsep=0pt]
  \item Le parentesi \textcolor{csPurple!80!black}{\textbf{delimitano liste}}, non raggruppano. Leggile come struttura.
  \item \fn{NIL} = \emph{falso} = \emph{lista vuota} = \emph{simbolo}. Triplice natura intenzionale.
  \item Nessun "return" implicito: il valore è quello dell'\textbf{ultima espressione}.
  \item Indentazione semanticamente irrilevante, ma \textbf{cruciale per leggibilità}.
  \item Errori nel REPL → \emph{debugger} → ispeziona → correggi → riprova.
  \item Funzione \textcolor{csPurple!80!black}{\textbf{prima}} degli argomenti: \fn{(+ 1 2)}, non \fn{1 + 2}.
\end{itemize}

\end{introbox}

\begin{tabular}{@{}c@{\quad}l@{\quad}l@{}}

\end{introbox}
