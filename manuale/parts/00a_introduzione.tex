\section*{Cosa rende Lisp diverso}
% =============================================================

\begin{introbox}[Principi fondamentali]{csBlue}

\begin{tabular}{@{}p{3.8cm}p{9.5cm}@{}}
\textbf{Codice = dati} & I programmi sono liste, la stessa struttura che manipolano. Puoi scrivere programmi che generano programmi.\\[6pt]
\textbf{Valutazione} & Non si "eseguono istruzioni": si valutano espressioni. Ogni espressione restituisce un valore.\\[6pt]
\textbf{Funzioni first-class} & Le funzioni sono valori: si passano, si restituiscono, si memorizzano.\\[6pt]
\textbf{Composizione} & Si costruisce componendo funzioni piccole, non accumulando passi.\\[6pt]
\textbf{Sintassi minima} & Poche regole, molte convenzioni. Le parentesi sono struttura, non rumore.\\[6pt]
\textbf{REPL} & Sviluppo interattivo: esplora, testa, costruisci incrementalmente.\\
\end{tabular}

\end{introbox}

\begin{introbox}[Imperativo vs Funzionale]{csOrange}

\renewcommand{\arraystretch}{1.25}
\centering
\begin{tabular}{@{}p{6.2cm}p{6.2cm}@{}}
\rowcolor{csOrange!10}
\textbf{Imperativo (C, Java, Python)} & \textbf{Lisp / Funzionale} \\
Istruzioni che modificano stato & Espressioni che producono valori \\
\rowcolor{csOrange!5}
Variabili = contenitori mutabili & Binding = nomi associati a valori \\
Cicli (\texttt{for}, \texttt{while}) & Ricorsione e higher-order \\
\rowcolor{csOrange!5}
Strutture di controllo fisse & Macro: controllo definibile \\
Sintassi specifica per costrutto & Sintassi uniforme (S-expression) \\
\rowcolor{csOrange!5}
Compilazione separata & Sviluppo interattivo nel REPL \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\end{introbox}

\begin{introbox}[Concetti chiave]{csGreen}

\begin{tabular}{@{}p{3cm}p{10.3cm}@{}}
\textbf{S-expression} & Atomo o lista di S-expression. Tutto il codice ha questa forma.\\[4pt]
\textbf{Valutazione} & Numeri/stringhe → se stessi. Simboli → loro valore. Liste → chiamata funzione.\\[4pt]
\textbf{Ambiente} & I binding visibili in un punto. \fn{let} crea ambienti locali.\\[4pt]
\textbf{Ricorsione} & Il modo naturale per elaborare liste e alberi.\\[4pt]
\textbf{Higher-order} & Funzioni che accettano/restituiscono funzioni: \fn{mapcar}, \fn{reduce}.\\[4pt]
\textbf{Macro} & Trasformazioni di codice a compile-time. Estendono il linguaggio.\\
\end{tabular}

\end{introbox}

\begin{introbox}[Per chi viene da altri linguaggi]{csPurple}

\begin{itemize}[leftmargin=1.2em, itemsep=3pt, topsep=0pt]
  \item Le parentesi delimitano liste, non raggruppano. Leggile come struttura.
  \item \fn{NIL} = falso = lista vuota = simbolo. Triplice natura intenzionale.
  \item Nessun "return" implicito: il valore è quello dell'ultima espressione.
  \item Indentazione semanticamente irrilevante, ma cruciale per leggibilità.
  \item Errori nel REPL → debugger → ispeziona → correggi → riprova.
  \item Funzione \emph{prima} degli argomenti: \fn{(+ 1 2)}, non \fn{1 + 2}.
\end{itemize}

\end{introbox}

\section*{Percorso di apprendimento}
% =============================================================

\begin{introbox}[Le 6 fasi]{csTeal}

\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}c@{\enspace}p{2.8cm}p{9.5cm}@{}}
\rowcolor{csTeal!10}
& \textbf{Fase} & \textbf{Obiettivi} \\
\textbf{1} & Fondamenta & REPL, \fn{defun}, \fn{if}/\fn{cond}, \fn{let}, \fn{dotimes} \\
\rowcolor{csTeal!5}
\textbf{2} & Liste & \fn{car}/\fn{cdr}/\fn{cons}, ricorsione su liste, pattern base \\
\textbf{3} & Ricorsione & Lineare → alberi, accumulatori, tail recursion \\
\rowcolor{csTeal!5}
\textbf{4} & Higher-order & \fn{mapcar}, \fn{reduce}, \fn{lambda}, closure \\
\textbf{5} & Strutture dati & Alist, plist, hash table, array, \fn{defstruct} \\
\rowcolor{csTeal!5}
\textbf{6} & Macro e progetti & Backquote, macro semplici, algoritmi, I/O, progetti \\
\end{tabular}
\renewcommand{\arraystretch}{1.0}

\end{introbox}
