% ================================================================
\section{Tipi di dato}
% ================================================================

\begin{concetto}{Tipizzazione dinamica}
\keyline{}{I tipi sono associati ai valori, non alle variabili.}

\detail{Una variabile è solo un nome che punta a un valore.
Il valore porta con sé il proprio tipo: non la variabile.}

\boxsep

\detail{Lo stesso nome può puntare prima a un numero, poi a una stringa,
poi a una lista: il linguaggio non si lamenta.}

\boxsep

\keyline{}{Il tipo viene controllato solo quando serve, a runtime.}

\detail{Non prima dell'esecuzione come in Java/C, ma nel momento
in cui il valore viene effettivamente usato.}
\end{concetto}

\begin{confronto}
\textbf{In Java/C} scrivi \texttt{int x = 5}: il compilatore sa che
\texttt{x} conterrà \emph{solo} interi e lo verifica prima di eseguire.
\textbf{In Common Lisp} non esiste questa dichiarazione.
Il prezzo: errori di tipo emergono a runtime.
Il vantaggio: meno cerimonia, più flessibilità, codice più conciso.
\end{confronto}

\begin{concetto}{\texttt{defparameter} e \texttt{setf}: variabili globali e assegnazione}
\keyline{\fn{defparameter}}{definisce una variabile globale e le assegna un valore iniziale.}
\keyline{\fn{setf}}{assegna un nuovo valore a una variabile (o a un posto qualsiasi).}

\begin{sintassi}
(defparameter *nome* valore)   \textit{; definisce e inizializza}\\
(setf *nome* nuovo-valore)     \textit{; riassegna}
\end{sintassi}

\detail{Per convenzione le variabili globali si scrivono con gli asterischi:
\texttt{*nome*}. Non è obbligatorio, ma è universale nel codice Lisp.
\fn{defparameter} crea sempre la variabile; \fn{setf} su una variabile
non ancora definita è un errore.}
\end{concetto}

\begin{esempio}[Tipizzazione dinamica in pratica]
\begin{lstlisting}
(defparameter *x* 42)
(type-of *x*)       ; => INTEGER  (o un sottotipo come FIXNUM)

(setf *x* "ciao")
(type-of *x*)       ; => (SIMPLE-ARRAY CHARACTER (4))

(setf *x* '(1 2 3))
(type-of *x*)       ; => CONS

;; Errore di tipo: emerge SOLO a runtime
(+ 3 "ciao")        ; ERRORE: "ciao" non e' un numero
\end{lstlisting}
\end{esempio}

\begin{nota}
Tipizzazione dinamica $\neq$ assenza di tipi.
I tipi ci sono: vengono controllati mentre il programma gira.
Per questo il REPL è fondamentale: testi ogni funzione con valori
reali e scopri subito eventuali errori di tipo.
\end{nota}

\seprule

\argomento{Numeri}

\begin{concetto}{Numeri: quattro sottotipi}
\boxitem{\texttt{integer}}{Interi a precisione arbitraria, limitati dalle risorse disponibili.}
\boxitem{\texttt{ratio}}{Frazioni esatte: \texttt{3/4}, \texttt{1/3}, \texttt{22/7}.}
\boxitem{\texttt{float}}{Virgola mobile; il formato concreto dipende dall'implementazione.}
\boxitem{\texttt{complex}}{Numeri complessi. Raramente usati.}

\boxsep
\detail{La distinzione più importante: \texttt{(/ 10 3)} restituisce
\texttt{10/3} (razionale esatto), non \texttt{3}. Le implementazioni moderne
(incluso Allegro CL) usano normalmente IEEE 754 per i float.}
\end{concetto}

\begin{esempio}[Aritmetica di base]
\begin{lstlisting}
(+ 3 4)          ; => 7       intero
(/ 10 3)         ; => 10/3    razionale esatto!
(/ 10.0 3)       ; => 3.3333  float
(* 1/3 3)        ; => 1       esatto

(expt 2 10)      ; => 1024
(sqrt 2)         ; => 1.4142...
(abs -7)         ; => 7
(mod 17 5)       ; => 2
(rem -7 3)       ; => -1      (mantiene segno del dividendo)

(floor   7/2)    ; => 3  (verso -inf)
(ceiling 7/2)    ; => 4  (verso +inf)
(truncate 7/2)   ; => 3  (taglia decimale)
(round   7/2)    ; => 4  (piu' vicino; 0.5 -> pari)
\end{lstlisting}
\end{esempio}

\begin{attenzione}
\textbf{\texttt{(/ 10 3)} non è la divisione intera.}
Restituisce \texttt{10/3} (razionale esatto), non \texttt{3}.
Per il quoziente intero: \texttt{(floor 10 3)} oppure
\texttt{(truncate 10 3)}, con due argomenti restituisce
quoziente e resto come valori multipli.
\end{attenzione}

\seprule

\argomento{Booleani, simboli, caratteri, stringhe}

\begin{esempio}[Gli altri tipi fondamentali]
\begin{lstlisting}
;; Booleani: solo nil e' falso, tutto il resto e' vero
t              ; => T
nil            ; => NIL  (falso, lista vuota, simbolo: tutto in uno)
(not nil)      ; => T
(not 0)        ; => NIL   (0 e' VERO)
(not "")       ; => NIL   (stringa vuota e' VERA)
(not '())      ; => T     ('() e' nil, quindi FALSO)

;; Simboli
'hello         ; => HELLO  (case-insensitive)
'Hello         ; => HELLO  (identico al precedente)

;; Caratteri
#\a            ; il carattere 'a'
#\Space        ; il carattere spazio
(char-code #\A)   ; => 65
(code-char 65)    ; => #\A

;; Stringhe
(length "ciao")              ; => 4
(string-upcase "ciao")       ; => "CIAO"
(concatenate 'string "a" "b" "c")  ; => "abc"
\end{lstlisting}
\end{esempio}

\begin{attenzione}
\textbf{Solo \texttt{nil} è falso.} Zero, stringa vuota, numero
negativo: tutto è vero. La lista vuota \texttt{()} è falsa
perché \textbf{è} \texttt{nil}: sono lo stesso oggetto.
Differenza critica rispetto a Python e C.
\end{attenzione}

\begin{intuizione}
\texttt{nil} è un unico oggetto con tre ruoli: valore falso,
lista vuota \texttt{()}, simbolo \texttt{NIL}.
Non tre cose diverse: la stessa identità.\\[4pt]
\texttt{(null nil)} $\Rightarrow$ \texttt{T} \quad
\texttt{(not nil)} $\Rightarrow$ \texttt{T} \quad
\texttt{(listp nil)} $\Rightarrow$ \texttt{T} \quad
\texttt{(equal nil '())} $\Rightarrow$ \texttt{T}
\end{intuizione}

\seprule

\argomento{Predicati di tipo}

\begin{sintassi}
(numberp x)   (integerp x)   (floatp x)    (stringp x)\\
(symbolp x)   (listp x)      (consp x)     (null x)\\
(functionp x) (atom x)       (characterp x)
\end{sintassi}

\begin{esempio}[Predicati di tipo]
\begin{lstlisting}
(numberp 42)       ; => T
(numberp "42")     ; => NIL
(atom 'x)          ; => T    (simbolo = atomo)
(atom '(a b))      ; => NIL  (lista = non atomo)
(consp '(a b))     ; => T    (lista non vuota)
(consp '())        ; => NIL  (nil non e' cons-cell)
\end{lstlisting}
\end{esempio}

\begin{concetto}{Uguaglianza: \fnt{=}, \fnt{eq}, \fnt{eql}, \fnt{equal}, \fnt{equalp}}
\keyline{\fn{=}}{confronta \emph{numeri}. Solo numeri.}
\keyline{\fn{eq}}{identità: stesso oggetto in memoria. Per simboli e \fn{nil}.}
\keyline{\fn{eql}}{come \fn{eq}, ma funziona anche per numeri e caratteri. Il default.}
\keyline{\fn{equal}}{confronto strutturale: liste, stringhe, array con stesso contenuto.}
\keyline{\fn{equalp}}{come \fn{equal}, ma ignora case nelle stringhe e confronta numeri di tipo diverso.}

\boxsep
\detail{Regola pratica: usa \fn{=} per i numeri, \fn{equal} per tutto il resto.
\fn{eql} è il default usato da \fn{member}, \fn{assoc}, \fn{case} e simili.
\fn{eq} solo quando sai di confrontare simboli o \fn{nil}.}
\end{concetto}

\section{Binding locale: \texttt{let} e \texttt{let*}}
% ================================================================

\begin{concetto}{\texttt{let} --- binding parallelo}
\keyline{}{Tutti i valori si calcolano \emph{prima} di legare qualsiasi variabile.}

\begin{sintassi}
(let ((var1 val1) (var2 val2) ...) corpo)
\end{sintassi}

\detail{I binding sono \emph{paralleli}: \texttt{var1} non è ancora
definita quando si calcola \texttt{val2}. Usalo quando i binding
sono indipendenti tra loro.}
\end{concetto}

\begin{concetto}{\texttt{let*} --- binding sequenziale}
\keyline{}{Ogni variabile è visibile nelle espressioni \emph{che seguono}.}

\begin{sintassi}
(let* ((var1 val1) (var2 val2) ...) corpo)
\end{sintassi}

\detail{Usalo quando un valore dipende da quello calcolato prima.}
\end{concetto}

\begin{confronto}
\textbf{Regola pratica:}
usa \fn{let} se i binding sono indipendenti tra loro,
usa \fn{let*} se uno dipende da un altro.
\end{confronto}

\begin{esempio}[\texttt{let} vs \texttt{let*}]
\begin{lstlisting}
;; let: binding indipendenti
(let ((x 10)
      (y 20))
  (+ x y))        ; => 30

;; let*: y usa x gia' definita
(let* ((x 10)
       (y (* x 2))
       (z (+ x y)))
  z)              ; => 30

;; let NON funzionerebbe qui:
;; (let ((x 10) (y (* x 2))) ...)  ERROR: x non ancora legata
\end{lstlisting}
\end{esempio}

\begin{esempio}[Pattern tipico: valori intermedi con \texttt{let*}]
\begin{lstlisting}
(defun ipotenusa (a b)
  (let* ((a2    (* a a))
         (b2    (* b b))
         (somma (+ a2 b2)))
    (sqrt somma)))

(ipotenusa 3 4)   ; => 5.0
\end{lstlisting}
\end{esempio}

\begin{attenzione}
\textbf{Le variabili di \texttt{let} non esistono fuori dal corpo.}
Sono strettamente locali. Non si dichiarano prima, non "escono".
In C/Java ogni variabile locale ha un blocco \texttt{\{\}} esplicito ---
in Lisp ogni \texttt{let} è quel blocco.
\end{attenzione}

\begin{workbook}
\textbf{Esercizi 4--5} (Ipotenusa, Distanza euclidea)\\[4pt]
Entrambi richiedono \fn{let*} per i valori intermedi.
L'esercizio~4 lo impone esplicitamente: è un esercizio
\emph{didattico} sul binding, non solo sull'aritmetica.
\end{workbook}

\section{Condizionali}
% ================================================================

\argomento{\texttt{if} --- la forma base}

\begin{concetto}{\texttt{if} è un'espressione, non un'istruzione}
\keyline{}{\fn{if} restituisce un valore --- il valore del ramo eseguito.}

\begin{sintassi}
(if condizione expr-vera expr-falsa)
\end{sintassi}

\detail{Il ramo falso è opzionale; se assente e la condizione è falsa,
il risultato è \texttt{NIL}.}

\boxsep
\detail{\textbf{Differenza da C/Java:} in Lisp \fn{if} è come l'operatore
ternario \texttt{cond ? a : b}, non uno statement. Puoi usarlo ovunque
ci si aspetti un valore.}
\end{concetto}

\begin{esempio}[\texttt{if} come espressione]
\begin{lstlisting}
(if (> 3 2) "maggiore" "minore")   ; => "maggiore"

;; Usabile dentro un'altra espressione
(+ 10 (if (oddp 7) 1 0))           ; => 11

;; Ramo falso assente
(if (> x 0) (print "positivo"))    ; => NIL se x <= 0
\end{lstlisting}
\end{esempio}

\begin{nota}
\fn{if} accetta \textbf{una sola espressione per ramo}.
Per eseguirne più d'una usa \fn{progn}:
\texttt{(if cond (progn e1 e2) altro)}.
In pratica, per corpi lunghi si preferisce \fn{when} o \fn{cond}.
\end{nota}

\seprule

\argomento{\texttt{when} / \texttt{unless} --- un solo caso}

\begin{sintassi}
(when   condizione corpo...)  \textit{; esegue se vero}\\
(unless condizione corpo...)  \textit{; esegue se falso}
\end{sintassi}

\begin{confronto}
Rispetto a \fn{if}: accettano \textbf{più espressioni} nel corpo
senza bisogno di \fn{progn}, ma non hanno ramo alternativo.
Usali per azioni con side-effect condizionati.
\end{confronto}

\begin{esempio}[\texttt{when} / \texttt{unless}]
\begin{lstlisting}
(when (> x 0)
  (print "positivo")
  (print "elaboro..."))    ; piu' espressioni nel corpo

(unless (null lst)
  (print (car lst)))
\end{lstlisting}
\end{esempio}

\seprule

\argomento{\texttt{cond} --- ramificazioni multiple}

\begin{concetto}{\texttt{cond}}
\keyline{}{Valuta i test in ordine; esegue il \emph{primo ramo vero}.}

\begin{sintassi}
(cond (test1 expr1...)
      (test2 expr2...)
      (t     default...))
\end{sintassi}

\detail{Restituisce il valore dell'ultima espressione del ramo eseguito.}
\detail{La clausola \texttt{(t ...)} è il default --- opzionale
ma sempre consigliata.}
\end{concetto}

\begin{esempio}[\texttt{cond} per ramificazioni multiple]
\begin{lstlisting}
(defun segno (n)
  (cond ((< n 0) 'negativo)
        ((= n 0) 'zero)
        (t       'positivo)))

(defun voto->giudizio (p)
  (cond ((>= p 90) "ottimo")
        ((>= p 75) "buono")
        ((>= p 60) "sufficiente")
        (t         "insufficiente")))
\end{lstlisting}
\end{esempio}

\seprule

\argomento{\texttt{case} --- dispatch su valori discreti}

\begin{sintassi}
(case valore
  (chiave1 expr...)
  (chiave2 expr...)
  (otherwise default...))
\end{sintassi}

\begin{esempio}[\texttt{case}]
\begin{lstlisting}
(defun calcola (op a b)
  (case op
    (+  (+ a b))
    (-  (- a b))
    (*  (* a b))
    (/  (if (= b 0) nil (/ a b)))
    (otherwise (error "operatore sconosciuto"))))
\end{lstlisting}
\end{esempio}

\begin{nota}
\fn{case} confronta con \fn{eql}: funziona con simboli e numeri,
\textbf{non con stringhe}. Per le stringhe usa \fn{cond} con \fn{equal}.
\end{nota}

\begin{intuizione}
\textbf{Quando usare cosa:}\\[4pt]
\begin{tabular}{@{}ll@{}}
\fn{if}             & 2 casi, espressioni brevi \\[2pt]
\fn{when}/\fn{unless} & 1 caso, corpo multi-espressione \\[2pt]
\fn{cond}           & 3+ casi, condizioni arbitrarie \\[2pt]
\fn{case}           & dispatch su valori discreti (simboli/numeri) \\
\end{tabular}
\end{intuizione}

\begin{workbook}
\textbf{Esercizi 8--16}
(Segno, Massimo, Triangolo, Bisestile, Giorni mese, Voto,
Terna, Radici, Calcolatrice)\\[4pt]
Ordine suggerito: \textbf{8, 11, 15, 16} per gli essenziali;
9, 10, 12, 13, 14 per completare.\\[4pt]
L'esercizio~15 (radici quadratiche) è il primo che combina
\fn{let*} e condizionali in modo non banale: vale la pena
farlo prima di andare avanti.
\end{workbook}

\section{Operatori logici: \texttt{and}, \texttt{or}, \texttt{not}}
% ================================================================

\begin{concetto}{\texttt{and} e \texttt{or} restituiscono valori, non solo T/NIL}
\keyline{\fn{and}}{valuta da sinistra; appena incontra \fn{nil} restituisce \fn{NIL}.
Se nessun valore è \fn{nil}, restituisce l'ultimo.}

\keyline{\fn{or}}{valuta da sinistra; appena incontra un valore non-\fn{nil} lo restituisce.
Se tutti sono \fn{nil}, restituisce \fn{NIL}.}

\boxsep
\detail{Entrambi usano la \textbf{valutazione cortocircuito}: le espressioni
dopo il punto di uscita non vengono nemmeno valutate.}
\end{concetto}

\begin{esempio}[Cosa restituiscono esattamente \texttt{and} e \texttt{or}]
\begin{lstlisting}
;; AND: restituisce NIL appena incontra nil, altrimenti l'ULTIMO valore
(and 1 2 3)        ; => 3      (nessun nil: restituisce l'ultimo)
(and 1 nil 3)      ; => NIL    (si ferma, il 3 non viene valutato)
(and nil 1 2)      ; => NIL    (si ferma subito)
(and)              ; => T      (zero argomenti: convenzione)

;; OR: restituisce il primo valore non-nil, altrimenti NIL
(or nil nil 42)    ; => 42     (42 e' non-nil: lo restituisce)
(or nil "ciao")    ; => "ciao"
(or nil nil nil)   ; => NIL
(or)               ; => NIL    (zero argomenti: convenzione)

;; Restituiscono il VALORE, non T/NIL
(and 10 20)        ; => 20     (non T!)
(or  nil 42)       ; => 42     (non T!)
\end{lstlisting}
\end{esempio}

\begin{confronto}
\textbf{Idioma 1 --- valore di default con \texttt{or}:}

\texttt{(or x default)} restituisce \texttt{x} se è non-nil,
altrimenti \texttt{default}. È il pattern \texttt{x ?? default}
di C\# o \texttt{x or default} di Python.

\begin{lstlisting}
(defun saluta (nome)
  (let ((n (or nome "Utente")))   ; nil -> usa "Utente"
    (format t "Ciao, ~a!~%" n)))

(saluta nil)    ; => Ciao, Utente!
(saluta "Ada")  ; => Ciao, Ada!
\end{lstlisting}

\medskip
\textbf{Idioma 2 --- guard clause con \texttt{and}:}

\texttt{(and condizione (f x))} esegue \texttt{(f x)} solo se
la condizione è vera; altrimenti restituisce \texttt{NIL}
senza chiamare \texttt{f}. È il pattern \texttt{cond \&\& f(x)} di C.

\begin{lstlisting}
(defun dividi (a b)
  (and (not (= b 0))   ; se b = 0, si ferma qui e restituisce NIL
       (/ a b)))       ; altrimenti calcola e restituisce il quoziente

(dividi 10 2)   ; => 5
(dividi 10 0)   ; => NIL  (non ha nemmeno tentato la divisione)
\end{lstlisting}
\end{confronto}

\section{Parametri avanzati e valori multipli}
% ================================================================

\argomento{\texttt{\&optional} --- parametri facoltativi}

\begin{concetto}{\texttt{\&optional} --- parametri facoltativi}
\keyline{}{Se il parametro \emph{non viene passato}, usa il valore di default.}

\begin{sintassi}
(defun f (obbligatorio \&optional (opt default)) corpo)
\end{sintassi}

\detail{Default omesso $\Rightarrow$ \texttt{NIL}.
Puoi avere più parametri opzionali; vengono riempiti in ordine.}
\end{concetto}

\begin{esempio}[\texttt{\&optional}]
\begin{lstlisting}
(defun saluta (&optional (nome "Mondo"))
  (format t "Ciao, ~a!~%" nome))

(saluta)        ; Ciao, Mondo!
(saluta "Ada")  ; Ciao, Ada!

(defun potenza (base &optional (esp 2))
  (expt base esp))

(potenza 3)     ; => 9   (3^2)
(potenza 3 4)   ; => 81  (3^4)
\end{lstlisting}
\end{esempio}

\seprule

\argomento{\texttt{\&rest} --- argomenti variabili}

\begin{concetto}{\texttt{\&rest} --- argomenti in numero variabile}
\keyline{}{Raccoglie \emph{tutti gli argomenti extra} in una lista.}

\begin{sintassi}
(defun f (a b \&rest resto) corpo)
\end{sintassi}

\detail{\texttt{resto} è una lista (vuota se non ci sono argomenti extra).
Usalo quando il numero di argomenti non è noto a priori.}
\end{concetto}

\begin{esempio}[\texttt{\&rest}]
\begin{lstlisting}
(defun somma-tutto (&rest numeri)
  (apply #'+ numeri))

(somma-tutto 1 2 3 4 5)   ; => 15
(somma-tutto)              ; => 0
\end{lstlisting}
\end{esempio}

\seprule

\argomento{Valori multipli}

\begin{concetto}{\texttt{values} e \texttt{multiple-value-bind}}
\keyline{}{Una funzione può restituire \emph{più valori} senza costruire una lista.}

\begin{sintassi}
(values v1 v2 ...)                        \textit{; lato produzione}\\
(multiple-value-bind (a b) (f) corpo)     \textit{; lato ricezione}
\end{sintassi}

\detail{Più efficiente di una lista: nessuna allocazione heap.
Se il chiamante vuole un solo valore, prende automaticamente il primo.}
\end{concetto}

\begin{esempio}[Valori multipli]
\begin{lstlisting}
(defun min-max (lst)
  (values (apply #'min lst)
          (apply #'max lst)))

(multiple-value-bind (minimo massimo)
    (min-max '(3 1 4 1 5 9))
  (format t "min=~a max=~a~%" minimo massimo))
; => min=1 max=9

;; floor con 2 argomenti: quoziente e resto
(multiple-value-bind (q r) (floor 17 5)
  (list q r))   ; => (3 2)
\end{lstlisting}
\end{esempio}

\begin{nota}
In un contesto che attende un solo valore (es. \texttt{(+ (floor 7 2) 1)}),
viene usato automaticamente solo il primo.
\end{nota}

\section{Iterazione di base}
% ================================================================

\begin{confronto}
La ricorsione è particolarmente naturale per attraversare strutture
ricorsive come liste e alberi. Per tutto il resto, Common Lisp offre
\fn{dotimes}, \fn{dolist} e \fn{loop}: costrutti potenti e flessibili,
usati normalmente in codice professionale.
\end{confronto}

\argomento{\texttt{dotimes} --- ripetizione N volte}

\begin{concetto}{\texttt{dotimes}}
\keyline{}{Ripete il corpo esattamente \emph{N volte}, con contatore da \fn{0} a \fn{N-1}.}

\begin{sintassi}
(dotimes (var N risultato) corpo)
\end{sintassi}

\detail{\texttt{risultato} è il valore restituito al termine (default \texttt{NIL}).
Omettilo se non ti serve il valore di ritorno.}
\end{concetto}

\begin{esempio}[\texttt{dotimes}]
\begin{lstlisting}
(dotimes (i 5)
  (format t "~a " i))
; stampa: 0 1 2 3 4

;; Con accumulatore e valore di ritorno
(let ((somma 0))
  (dotimes (i 10 somma)
    (setf somma (+ somma i))))
; => 45
\end{lstlisting}
\end{esempio}

\seprule

\argomento{\texttt{dolist} --- iterare su una lista}

\begin{sintassi}
(dolist (var lista risultato) corpo)
\end{sintassi}

\begin{esempio}[\texttt{dolist}]
\begin{lstlisting}
(dolist (x '(1 2 3 4 5))
  (format t "~a " (* x x)))
; stampa: 1 4 9 16 25
\end{lstlisting}
\end{esempio}

\seprule

\argomento{\texttt{loop} --- iterazione flessibile}

\begin{confronto}
\fn{loop} è il costrutto iterativo più potente di Common Lisp.
Ha una mini-sintassi propria in inglese. Qui vediamo le forme
essenziali; per un trattamento completo si rimanda alla documentazione
ufficiale o al Practical Common Lisp di Seibel.
\end{confronto}

\begin{esempio}[\texttt{loop} --- forme essenziali]
\begin{lstlisting}
;; Contatore con output
(loop for i from 1 to 5
      do (format t "~a " i))
; stampa: 1 2 3 4 5

;; Raccogliere risultati in una lista
(loop for i from 1 to 5
      collect (* i i))
; => (1 4 9 16 25)

;; Somma con accumulatore
(loop for i from 1 to 100
      sum i)
; => 5050

;; Iterare su una lista
(loop for x in '(a b c)
      do (print x))
\end{lstlisting}
\end{esempio}

\begin{attenzione}
\textbf{Quando usare la ricorsione e quando \fn{loop}.}
Negli esempi con \fn{dotimes} compare \fn{setf}: è normale,
i costrutti iterativi lavorano spesso con stato locale.
Regola pratica: usa la ricorsione quando la struttura del problema
è ricorsiva (lista, albero); usa \fn{dotimes}/\fn{loop} per iterazioni
su range numerici o quando \fn{loop \ldots\ collect}/\fn{sum}/\fn{count}
rende l'intento più diretto.
\end{attenzione}

\section{Output con \texttt{format}}
% ================================================================

\begin{concetto}{\texttt{format} --- output formattato}
\keyline{}{Stampa su un destinatario con una \emph{stringa di controllo}.}

\begin{sintassi}
(format destinazione stringa-controllo argomenti...)
\end{sintassi}

\detail{\texttt{t} $\Rightarrow$ standard output.\quad
\texttt{nil} $\Rightarrow$ non stampa, restituisce una stringa.}
\end{concetto}

\begin{esempio}[Direttive essenziali]
\begin{lstlisting}
(format t "Ciao, ~a!~%"  "Ada")  ; ~a = qualsiasi valore, ~% = newline
(format t "Valore: ~s~%" '(1 2)) ; ~s = rappresentazione sorgente
(format t "Numero: ~d~%" 42)     ; ~d = intero decimale
(format t "Float:  ~f~%" 3.14)   ; ~f = virgola mobile
(format t "~&")                  ; ~& = newline se non gia' a inizio riga
(format t "~10d~%" 42)           ; ~10d = campo di 10 caratteri

;; Costruire una stringa senza stampare
(format nil "val=~a" 42)         ; => "val=42"
\end{lstlisting}
\end{esempio}

\textbf{Quello che hai imparato:}

\medskip
\begin{itemize}
  \item \textbf{Tipi}: interi senza overflow, razionali esatti, float,
        simboli, stringhe, caratteri. Solo \texttt{NIL} è falso.
  \item \textbf{\texttt{let}/\texttt{let*}}: binding locale parallelo
        vs sequenziale. Le variabili non escono dal corpo.
  \item \textbf{Condizionali}: \fn{if} (2 casi),
        \fn{when}/\fn{unless} (1 caso), \fn{cond} ($n$ casi),
        \fn{case} (dispatch su valori discreti).
  \item \textbf{\texttt{and}/\texttt{or}}: short-circuit con valori reali.
        Idiomi: default con \fn{or}, guard con \fn{and}.
  \item \textbf{Parametri}: \fn{\&optional} (con default),
        \fn{\&rest} (argomenti variabili).
        Valori multipli con \fn{values}/\fn{multiple-value-bind}.
  \item \textbf{Iterazione}: \fn{dotimes}, \fn{dolist}, \fn{loop}
        per iterazioni imperative; ricorsione per strutture ricorsive.
  \item \textbf{\fn{format}}: \texttt{\textasciitilde a},
        \texttt{\textasciitilde d}, \texttt{\textasciitilde f},
        \texttt{\textasciitilde \%}, \texttt{\textasciitilde s}.
\end{itemize}
